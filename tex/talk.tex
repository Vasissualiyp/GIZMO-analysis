\begin{frame}{Motivation}
\begin{itemize}
  \item Pop~III stars set the IMF of the first generation
        $\to$ chemical enrichment, reionization, SMBH seeds
  \vspace{0.3em}
  \item \textbf{Key open question:} Do Pop~III accretion disks fragment?
        \begin{itemize}
          \item If yes $\to$ multiple lower-mass Pop~III stars
          \item If no $\to$ single very massive star
        \end{itemize}
  \vspace{0.3em}
  \item Previous work:
        \begin{itemize}
          \item Isolated collapse simulations --- no cosmological context
          \item Cosmological zoom-ins --- under-resolved at disk scale
        \end{itemize}
  \vspace{0.3em}
  \item \textbf{This work:} First cosmological zoom-in with
        Jeans-resolved protostellar disk at $z \sim 21$
\end{itemize}
\end{frame}

\begin{frame}{Method: cutouts \& disk identification}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{Spatial cutouts}
\begin{itemize}
  \item $r = 0.005\kpc/h \approx 330\AU$ around most massive sink
  \item Keeps only Jeans-refined gas + sinks
  \item Removes 88M low-res CDM particles
\end{itemize}
\vspace{0.5em}
\textbf{Disk identification}
\begin{enumerate}\small
  \item Centre on most massive sink
  \item COM velocity from gas within $r_{\rm search}$
  \item Disk normal $\hat{L}$ from angular momentum
  \item Geometric: $r_{\rm cyl} < r_{\rm max}$, $|z|/r < 0.3$,
        $\rho > 10^{-15}\,\mathrm{g/cm^3}$
  \item Kinematic: $v_\phi > 0$, $v_\phi / v_K > 0.3$
\end{enumerate}
\end{column}
\begin{column}{0.45\textwidth}
\includegraphics[width=\textwidth]{\fcut/frame_density_0050.png}
\end{column}

\begin{frame}{Summary}
\begin{enumerate}
  \item First cosmological zoom-in resolving Pop~III disk fragmentation
        at $z \sim 21$
  \vspace{0.4em}
  \item Disk forms, goes Toomre-unstable at $t \sim 3.6\kyr$,
        fragments via spiral arm
  \vspace{0.4em}
  \item $M_* \propto t^2$, SFE $\sim 50$--$80\%$
  \vspace{0.4em}
  \item Sink formation produces outward-propagating shocks
        (visible in $\sigma_r$ heatmaps)
  \vspace{0.4em}
  \item Burgers turbulence ($k^{-2}$) throughout the disk
  \vspace{0.4em}
  \item[\textbf{$\Rightarrow$}] \textbf{Pop~III accretion disks do
        fragment} $\to$ multiple Pop~III stars, not just one
\end{enumerate}
\end{frame}
